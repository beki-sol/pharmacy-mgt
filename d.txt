-- 7. Batches for each drug in each branch (2 batches per drug per branch)
DO $$
DECLARE
  drug_rec RECORD;
  branch_rec RECORD;
  batch_num INT;
  qty_per_batch INT;
BEGIN
  FOR drug_rec IN SELECT id, name FROM "Drug" LOOP
    FOR branch_rec IN SELECT id, name FROM "Branch" LOOP
      SELECT stock INTO qty_per_batch 
      FROM "DrugBranch" 
      WHERE "drugId" = drug_rec.id AND "branchId" = branch_rec.id;
      IF qty_per_batch IS NULL THEN
        CONTINUE;
      END IF;
      FOR batch_num IN 1..2 LOOP
        INSERT INTO "DrugBatch" (
          id, "drugId", "branchId", "batchNumber", "expiryDate", quantity, remaining, "costPrice", "createdAt"
        ) VALUES (
          gen_random_uuid()::text,
          drug_rec.id,
          branch_rec.id,
          'BATCH-' || branch_rec.name || '-' || drug_rec.name || '-' || batch_num,
          NOW() + (CASE WHEN batch_num = 1 THEN INTERVAL '2 years' ELSE INTERVAL '1 year' END),
          qty_per_batch / 2,
          qty_per_batch / 2,
          (SELECT "costPrice" FROM "Drug" WHERE id = drug_rec.id),
          NOW()
        );
      END LOOP;
    END LOOP;
  END LOOP;
END $$;

-- 11. Update DrugBranch stock after purchase orders (increase stock)
UPDATE "DrugBranch" db
SET stock = db.stock + sub.total_added
FROM (
  SELECT po."branchId", pi."drugId", SUM(pi.quantity) AS total_added
  FROM "PurchaseOrder" po
  JOIN "PurchaseItem" pi ON pi."purchaseOrderId" = po.id
  WHERE po.status = 'DELIVERED'
  GROUP BY po."branchId", pi."drugId"
) sub
WHERE db."branchId" = sub."branchId" AND db."drugId" = sub."drugId";

-- 13. Sale Items (for completed sales, pick random drugs from the same branch)
WITH completed_sales AS (
  SELECT id, "branchId" FROM "Sale" WHERE status = 'COMPLETED'
)
INSERT INTO "SaleItem" (
  id, "saleId", "drugId", quantity, "unitPrice", discount, tax, subtotal, "batchId", "batchNumber"
)
SELECT
  gen_random_uuid()::text,
  s.id,
  d.id,
  floor(random() * 5 + 1)::int,
  d.price,
  0,
  0,
  0, -- will update later
  (SELECT id FROM "DrugBatch" WHERE "drugId" = d.id AND "branchId" = s."branchId" LIMIT 1),
  NULL
FROM completed_sales s
CROSS JOIN LATERAL (
  SELECT id, price, name FROM "Drug" d WHERE d.id IN (SELECT "drugId" FROM "DrugBranch" WHERE "branchId" = s."branchId")
  ORDER BY random() LIMIT floor(random() * 3 + 1)::int
) d;

-- 14. Update sale items subtotal and update sale totals
UPDATE "SaleItem" si
SET subtotal = si.quantity * si."unitPrice" - si.discount;

UPDATE "Sale" s
SET "totalAmount" = (
    SELECT COALESCE(SUM(subtotal), 0) FROM "SaleItem" WHERE "saleId" = s.id
  ),
  "netAmount" = (
    SELECT COALESCE(SUM(subtotal), 0) FROM "SaleItem" WHERE "saleId" = s.id
  ) - s.discount + s.tax
WHERE s.status = 'COMPLETED';

-- 15. Inventory Logs for sales (SALE)

INSERT INTO "InventoryLog" (
  id, "drugId", "branchId", type, quantity, "previousStock", "newStock", "saleId", "batchNumber", "userId", "createdAt"
)
SELECT
  gen_random_uuid()::text,
  si."drugId",
  s."branchId",
  'SALE'::"InventoryAction",
  -si.quantity,
  db.stock + si.quantity,
  db.stock,
  s.id,
  si."batchNumber",
  (SELECT id FROM "User" WHERE email = 'admin@pharmacy.com'),
  s."createdAt"
FROM "Sale" s
JOIN "SaleItem" si ON si."saleId" = s.id
JOIN "DrugBranch" db ON db."drugId" = si."drugId" AND db."branchId" = s."branchId"
WHERE s.status = 'COMPLETED';

-- 16. Update DrugBranch stock after sales (decrease)
UPDATE "DrugBranch" db
SET stock = db.stock - sub.total_sold
FROM (
  SELECT s."branchId", si."drugId", SUM(si.quantity) AS total_sold
  FROM "Sale" s
  JOIN "SaleItem" si ON si."saleId" = s.id
  WHERE s.status = 'COMPLETED'
  GROUP BY s."branchId", si."drugId"
) sub
WHERE db."branchId" = sub."branchId" AND db."drugId" = sub."drugId";

-- 17. Update batch remaining after sales
UPDATE "DrugBatch" b
SET remaining = b.remaining - sub.sold_qty
FROM (
  SELECT si."batchId", SUM(si.quantity) AS sold_qty
  FROM "SaleItem" si
  JOIN "Sale" s ON si."saleId" = s.id
  WHERE s.status = 'COMPLETED' AND si."batchId" IS NOT NULL
  GROUP BY si."batchId"
) sub
WHERE b.id = sub."batchId";
